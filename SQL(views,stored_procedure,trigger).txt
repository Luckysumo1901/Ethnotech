create table employee (e_id int primary key auto_increment,e_name varchar(20),salary int);
insert into employee value
(1,"sai",10000),
(2,"sam",50000),
(3,"sana",20000);

create table employee_msg(
message varchar(100)
);

delimiter //
create trigger employee_insert_trigger
after insert 
on employee
for each row
begin
insert into employee_msg(message)
	values(
		concat('Employee Added: ',new.e_name)
	);
end //
delimiter ;

insert into employee(e_name,salary)
values("Rahul",30000);

select * from employee_msg;

delimiter //
create trigger emp_after_update
after update
on employee
for each row
begin
insert into employee_msg(message)
	values(
		concat(
			new.e_name,'salary changed from ',
            old.salary,'to',
            new.salary
        )
	);
end //
delimiter ;

update employee set salary=50000 where e_id=1;

select * from employee_msg;

delimiter //
create trigger emp_after_delete
after delete
on employee
for each row
begin
insert into employee_msg(message)
	values(
		concat(
			'deleted employee id ',old.e_id
        )
	);
end //
delimiter ;

delete from employee where e_id=1;

select * from employee_msg;
